import { createClient } from "npm:@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "https://asinetworkfrl.com", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders, }); } try { const authHeader = req.headers.get("Authorization"); if (!authHeader?.startsWith("Bearer ")) { return new Response( JSON.stringify({ error: "Unauthorized", }), { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } const accessToken = authHeader.replace("Bearer ", ""); const supabaseUrl = Deno.env.get("SUPABASE_URL"); const supabaseSecret = Deno.env.get("ASI_SUPABASE_SECRET_KEY"); const muxTokenId = Deno.env.get("MUX_ACCESS_TOKEN_ID"); const muxTokenSecret = Deno.env.get("MUX_ACCESS_TOKEN_SECRET"); if ( !supabaseUrl || !supabaseSecret || !muxTokenId || !muxTokenSecret ) { throw new Error( "Required server secrets are missing." ); } const supabaseAdmin = createClient( supabaseUrl, supabaseSecret ); /* ========================= VERIFY USER ========================= */ const { data: userData, error: userError, } = await supabaseAdmin.auth.getUser( accessToken ); if ( userError || !userData?.user ) { return new Response( JSON.stringify({ error: "Invalid login session.", }), { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } const user = userData.user; /* ========================= VERIFY ADMIN ========================= */ const { data: admin, error: adminError, } = await supabaseAdmin .from("admins") .select("user_id") .eq("user_id", user.id) .maybeSingle(); if ( adminError || !admin ) { return new Response( JSON.stringify({ error: "Administrator access required.", }), { status: 403, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } /* ========================= READ REQUEST ========================= */ const body = await req.json(); const episodeId = body?.episodeId; const title = body?.title || "ASI Network FRL Episode"; if (!episodeId) { return new Response( JSON.stringify({ error: "Episode ID is required.", }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } /* ========================= VERIFY EPISODE EXISTS ========================= */ const { data: episode, error: episodeError, } = await supabaseAdmin .from("episodes") .select(` id, episode_number, title, video_id `) .eq("id", episodeId) .maybeSingle(); if ( episodeError || !episode ) { return new Response( JSON.stringify({ error: "Episode was not found.", }), { status: 404, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } /* ========================= CREATE MUX DIRECT UPLOAD ========================= */ const basicAuth = btoa( `${muxTokenId}:${muxTokenSecret}` ); const muxResponse = await fetch( "https://api.mux.com/video/v1/uploads", { method: "POST", headers: { "Authorization": `Basic ${basicAuth}`, "Content-Type": "application/json", }, body: JSON.stringify({ cors_origin: "https://asinetworkfrl.com", timeout: 7200, new_asset_settings: { playback_policies: [ "signed" ], passthrough: episodeId, meta: { title: title, external_id: episodeId, }, }, }), } ); const muxResult = await muxResponse.json(); if (!muxResponse.ok) { console.error( "Mux upload creation failed:", muxResult ); return new Response( JSON.stringify({ error: "Mux could not create the upload.", }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } /* ========================= RETURN TEMPORARY UPLOAD URL ========================= */ return new Response( JSON.stringify({ success: true, uploadId: muxResult.data.id, uploadUrl: muxResult.data.url, episodeId: episodeId, }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } catch (error) { console.error( "ASI Mux upload error:", error ); return new Response( JSON.stringify({ error: error instanceof Error ? error.message : "Unexpected upload error.", }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } });
